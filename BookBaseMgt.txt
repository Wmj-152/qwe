<script>
const base = "";
let currentType = "on"; // on=在架 off=已下架
let currentPage = 1;
const pageSize = 6;

// 标签点击切换
document.querySelectorAll(".tab-item").forEach(tab=>{
    tab.onclick = function(){
        document.querySelectorAll(".tab-item").forEach(i=>i.classList.remove("active"));
        this.classList.add("active");
        currentType = this.dataset.type;
        currentPage = 1;
        renderList();
    }
})

function getForm(){
    return {
        bookId: document.getElementById("bookId").value.trim(),
        bookName: document.getElementById("bookName").value.trim(),
        author: document.getElementById("author").value.trim(),
        publisher: document.getElementById("publisher").value.trim(),
        stock: document.getElementById("stock").value
    }
}

// 软删除（下架）
async function softDelete(){
    let id = document.getElementById("bookId").value.trim();
    if(!id){
        alert("请先填入图书编号！");
        return;
    }
    if(!confirm("确定下架这本书？下架后可以在【已下架图书】里恢复")) return;
    let res = await fetch(base+"/soft_delete",{
        method:"POST",
        headers:{"Content-Type":"application/json"},
        body:JSON.stringify({bookId:id})
    })
    let json = await res.json();
    alert(json.msg);
    renderList();
}

// 恢复上架
async function restoreBook(bookId){
    if(!confirm("确定恢复上架？")) return;
    let res = await fetch(base+"/restore",{
        method:"POST",
        headers:{"Content-Type":"application/json"},
        body:JSON.stringify({bookId:bookId})
    })
    let json = await res.json();
    alert(json.msg);
    renderList();
}

// 彻底物理删除
async function hardDelete(bookId){
    if(!confirm("⚠️ 警告：这是彻底删除，数据无法找回！确认吗？")) return;
    let res = await fetch(base+"/hard_delete",{
        method:"POST",
        headers:{"Content-Type":"application/json"},
        body:JSON.stringify({bookId:bookId})
    })
    let json = await res.json();
    alert(json.msg);
    renderList();
}

async function addBook(){
    let obj = getForm();
    let res = await fetch(base+"/add",{
        method:"POST",
        headers:{"Content-Type":"application/json"},
        body:JSON.stringify(obj)
    })
    let json = await res.json();
    alert(json.msg);
    currentType = "on";
    document.querySelector('[data-type="on"]').click();
}

async function queryById(){
    let id = document.getElementById("bookId").value.trim();
    if(!id){alert("请输入图书编号");return;}
    let res = await fetch(base+"/get?bookId="+encodeURIComponent(id));
    let json = await res.json();
    if(!json.data){
        alert(json.msg);
        return;
    }
    document.getElementById("bookName").value = json.data.bookName;
    document.getElementById("author").value = json.data.author;
    document.getElementById("publisher").value = json.data.publisher;
    document.getElementById("stock").value = json.data.stock;
}

async function updateBook(){
    let obj = getForm();
    let res = await fetch(base+"/update",{
        method:"POST",
        headers:{"Content-Type":"application/json"},
        body:JSON.stringify(obj)
    })
    let json = await res.json();
    alert(json.msg);
    renderList();
}

async function search(){
    currentPage = 1;
    renderList();
}

async function resetAll(){
    document.getElementById("searchKey").value = "";
    currentPage = 1;
    renderList();
}

// 核心渲染：分页+标签筛选
async function renderList(){
    let kw = document.getElementById("searchKey").value.trim();
    let res = await fetch(base+`/search?keyword=${encodeURIComponent(kw)}&type=${currentType}&page=${currentPage}&size=${pageSize}`);
    let json = await res.json();
    let list = json.list;
    let total = json.total;
    let tb = document.getElementById("tb");
    tb.innerHTML="";
    list.forEach(item=>{
        let tr = document.createElement("tr");
        let opHtml = "";
        if(currentType === "on"){
            opHtml = `<button class="btn-danger" onclick="softDeleteById('${item.bookId}')">下架</button>`
        }else{
            opHtml = `<button class="btn-success" onclick="restoreBook('${item.bookId}')">恢复上架</button>
                      <button class="btn-warning" onclick="hardDelete('${item.bookId}')">彻底删除</button>`
        }
        tr.innerHTML=`<td>${item.bookId}</td>
                      <td>${item.bookName}</td>
                      <td>${item.author}</td>
                      <td>${item.publisher}</td>
                      <td>${item.stock}</td>
                      <td>${opHtml}</td>`;
        tb.appendChild(tr);
    })
    // 分页按钮
    let pageBox = document.getElementById("pageBox");
    let totalPage = Math.ceil(total / pageSize);
    let pageHtml = `共 ${total} 条 &nbsp;&nbsp;`;
    if(currentPage>1){
        pageHtml += `<button onclick="goPage(${currentPage-1})">上一页</button>`
    }
    pageHtml += ` 第 ${currentPage} / ${totalPage||1} 页 `
    if(currentPage < totalPage){
        pageHtml += `<button onclick="goPage(${currentPage+1})">下一页</button>`
    }
    pageBox.innerHTML = pageHtml;
}
function goPage(p){
    currentPage = p;
    renderList();
}
// 单独给表格行的下架按钮

        method:"POST",
        headers:{"Content-Type":"application/json"},
        body:JSON.stringify({bookId:bookId})
    })
    let json = await res.json();
    alert(json.msg);
    renderList();
}
