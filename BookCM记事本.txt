var editCid = null;   // 当前正在修改的分类编号，null 表示新增状态

function showMsg(text, ok) {
    var el = document.getElementById("msg");
    el.innerText = text;
    el.style.color = ok ? "#2a7" : "#c33";
}

// ========== 列表 / 查询 ==========
function loadList() {
    var kw = document.getElementById("kw").value.trim();
    fetch("/api/categories?kw=" + encodeURIComponent(kw))
        .then(function (r) { return r.json(); })
        .then(function (res) {
            var html = "";
            if (res.data.length == 0) {
                html = '<tr><td colspan="5" class="center">没有符合条件的分类</td></tr>';
            } else {
                res.data.forEach(function (c) {
                    var refCell = c.ref > 0
                        ? "<button class='ref-link' onclick=\"showBooks('" + c.cid + "')\">" + c.ref + "</button>"
                        : "<span class='ref-none'>0</span>";
                    html += "<tr>" +
                        "<td>" + c.cid + "</td>" +
                        "<td>" + c.cname + "</td>" +
                        "<td>" + (c.remark || "") + "</td>" +
                        "<td class='center'>" + refCell + "</td>" +
                        "<td>" +
                        "<button onclick=\"startEdit('" + c.cid + "')\">编辑</button>" +
                        "<button onclick=\"delCategory('" + c.cid + "','" + c.cname + "')\">删除</button>" +
                        "</td>" +
                        "</tr>";
                });
            }
            document.getElementById("tbody").innerHTML = html;
        });
}

function resetQuery() {
    document.getElementById("kw").value = "";
    loadList();
}

// ========== 引用明细：点引用数看是哪些书 ==========
function showBooks(cid) {
    fetch("/api/categories/" + encodeURIComponent(cid) + "/books")
        .then(function (r) { return r.json(); })
        .then(function (res) {
            if (res.code != 0) { showMsg(res.msg, false); return; }
            var d = res.data;
            document.getElementById("dlgTitle").innerText =
                "分类 " + d.cid + "（" + d.cname + "）的引用图书";

            var html = "<div>共 " + d.books.length + " 本图书引用该分类：</div>";
            if (d.books.length > 0) {
                html += "<table><tr><th>图书编号</th><th>书名</th></tr>";
                d.books.forEach(function (b) {
                    html += "<tr><td>" + b.bid + "</td><td>" + b.title + "</td></tr>";
                });
                html += "</table>";
            }
            html += "<div style='margin-top:12px;color:#888'>该分类被图书引用期间不允许删除。</div>";
            document.getElementById("dlgBody").innerHTML = html;
            document.getElementById("mask").className = "mask show";
        });
}

function closeBooks(e) {
    document.getElementById("mask").className = "mask";
}

// ========== 新增 / 修改 保存 ==========
function saveForm() {
    var cid = document.getElementById("cid").value.trim();
    var cname = document.getElementById("cname").value.trim();
    var remark = document.getElementById("remark").value.trim();

    if (editCid) {
        // 修改保存
        fetch("/api/categories/" + encodeURIComponent(editCid), {
            method: "PUT",
            headers: {"Content-Type": "application/json"},
            body: JSON.stringify({cname: cname, remark: remark})
        }).then(function (r) { return r.json(); }).then(function (res) {
            showMsg(res.msg, res.code == 0);
            if (res.code == 0) { cancelEdit(); loadList(); }
        });
    } else {
        // 新增
        if (!cid) { showMsg("分类编号不能为空！", false); return; }
        fetch("/api/categories", {
            method: "POST",
            headers: {"Content-Type": "application/json"},
            body: JSON.stringify({cid: cid, cname: cname, remark: remark})
        }).then(function (r) { return r.json(); }).then(function (res) {
            showMsg(res.msg, res.code == 0);
            if (res.code == 0) {
                document.getElementById("cid").value = "";
                document.getElementById("cname").value = "";
                document.getElementById("remark").value = "";
                loadList();
            }
        });
    }
}

// ========== 编辑：按编号回显 ==========
function startEdit(cid) {
    fetch("/api/categories/" + encodeURIComponent(cid))
        .then(function (r) { return r.json(); })
        .then(function (res) {
            if (res.code != 0) { showMsg(res.msg, false); return; }
            var c = res.data;
            editCid = c.cid;
            document.getElementById("cid").value = c.cid;
            document.getElementById("cid").disabled = true;
            document.getElementById("cname").value = c.cname;
            document.getElementById("remark").value = c.remark;
            document.getElementById("btnSave").innerText = "保存修改";
            document.getElementById("btnCancel").style.display = "inline-block";

            var tip = document.getElementById("editTip");
            tip.style.display = "block";
            tip.innerText = "正在修改分类 " + c.cid + "（" + c.cname + "），编号不可修改，改完点“保存修改”。";
            showMsg("已按编号回显该分类信息。", true);
        });
}

function cancelEdit() {
    editCid = null;
    document.getElementById("cid").value = "";
    document.getElementById("cid").disabled = false;
    document.getElementById("cname").value = "";
    document.getElementById("remark").value = "";
    document.getElementById("btnSave").innerText = "新增分类";
    document.getElementById("btnCancel").style.display = "none";
    document.getElementById("editTip").style.display = "none";
}


// ========== 删除 ==========
function delCategory(cid, cname) {
    if (!confirm("确定删除分类「" + cname + "」吗？")) return;
    fetch("/api/categories/" + encodeURIComponent(cid), {method: "DELETE"})
        .then(function (r) { return r.json(); })
        .then(function (res) {
            showMsg(res.msg, res.code == 0);
            if (res.code == 0) {
                if (editCid === cid) cancelEdit();
                loadList();
            }
        });
}
