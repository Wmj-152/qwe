import json
from urllib.parse import urlparse, parse_qs

    {
        "bookId": "B001",
        "bookName": "西游记",
        "author": "吴承恩",
        "publisher": "人民文学出版社",

        "is_deleted": False
    },
    {
        "bookId": "B002",

    {

        "stock": 15,
        "is_deleted": False
    }
]

class MyHandler(BaseHTTPRequestHandler):
    def _set_headers(self, code=200):
        self.send_response(code)
        self.send_header("Content-Type", "application/json;charset=utf-8")
        self.end_headers()

    def do_GET(self):
        if self.path == "/":
            self.send_response(200)
            self.send_header("Content-Type", "text/html;charset=utf-8")
            self.end_headers()
            with open("index.html", "r", encoding="utf-8") as f:
                self.wfile.write(f.read().encode("utf-8"))
            return

        parsed = urlparse(self.path)
        qs = parse_qs(parsed.query)

        # 根据编号查询单本
        if parsed.path == "/get":
            bid = qs.get("bookId", [""])[0]
            res = {"data": None, "msg": "未找到该图书"}
            for b in book_db:
                if b["bookId"] == bid:
                    res["data"] = b
                    break
            self._set_headers()
            self.wfile.write(json.dumps(res, ensure_ascii=False).encode("utf-8"))
            return

        # 分页+筛选：type=on  在架 / type=off 已下架；page页码，size每页数量
        if parsed.path == "/search":
            kw = qs.get("keyword", [""])[0].lower()
            show_type = qs.get("type", ["on"])[0]
            page = int(qs.get("page", ["1"])[0])
            page_size = int(qs.get("size", ["6"])[0])

            # 先过滤在架/下架
            if show_type == "on":
                filter_list = [b for b in book_db if b["is_deleted"] is False]
            else:
                filter_list = [b for b in book_db if b["is_deleted"] is True]
            # 书名模糊搜索
            if kw != "":
                filter_list = [b for b in filter_list if kw in b["bookName"].lower()]
            # 分页切片
            total = len(filter_list)
            start = (page - 1) * page_size
            end = start + page_size
            page_data = filter_list[start:end]
            ret = {
                "list": page_data,
                "total": total,
                "page": page,
                "page_size": page_size
            }
            self._set_headers()
            self.wfile.write(json.dumps(ret, ensure_ascii=False).encode("utf-8"))
            return

        self._set_headers(404)
        self.wfile.write(json.dumps({"msg": "路径不存在"}, ensure_ascii=False).encode("utf-8"))

    def do_POST(self):
        length = int(self.headers["Content-Length"])
        body = self.rfile.read(length).decode("utf-8")
        data = json.loads(body)

        # 新增图书，默认在架 is_deleted=False
        if self.path == "/add":
            book_id = data.get("bookId", "").strip()
            book_name = data.get("bookName", "").strip()
            author = data.get("author", "").strip()
            publisher = data.get("publisher", "").strip()
            stock = data.get("stock")

            if not book_id:
                msg = "新增失败：图书编号不能为空"
            elif not book_name:
                msg = "新增失败：书名不能为空"
            elif not author:
                msg = "新增失败：作者不能为空"
            elif not publisher:
                msg = "新增失败：出版社不能为空"
            elif stock is None or int(stock) <= 0:
                msg = "新增失败：库存必须大于0"
            elif any(b["bookId"] == book_id for b in book_db):
                msg = "新增失败：该图书编号已存在"
            else:
                new_book = {
                    "bookId": book_id,
                    "bookName": book_name,
                    "author": author,
                    "publisher": publisher,
                    "stock": int(stock),
                    "is_deleted": False
                }
                book_db.append(new_book)
                msg = "新增成功"
            self._set_headers()
            self.wfile.write(json.dumps({"msg": msg}, ensure_ascii=False).encode("utf-8"))
            return

        # 修改图书
        if self.path == "/update":
            book_id = data.get("bookId", "").strip()
            book_name = data.get("bookName", "").strip()
            author = data.get("author", "").strip()
            publisher = data.get("publisher", "").strip()
            stock = data.get("stock")

            if not book_id:
                msg = "修改失败：图书编号不能为空"
            elif not book_name:
                msg = "修改失败：书名不能为空"
            elif not author:
                msg = "修改失败：作者不能为空"
            elif not publisher:
                msg = "修改失败：出版社不能为空"
            elif stock is None or int(stock) <= 0:
                msg = "修改失败：库存必须大于0"
            else:
                find = False
                for b in book_db:
                    if b["bookId"] == book_id:
                        b["bookName"] = book_name
                        b["author"] = author
                        b["publisher"] = publisher
                        b["stock"] = int(stock)
                        find = True
                        break
                if find:
                    msg = "修改保存成功"
                else:
                    msg = "修改失败：不存在该编号图书"
            self._set_headers()
            self.wfile.write(json.dumps({"msg": msg}, ensure_ascii=False).encode("utf-8"))
            return

        # 软删除：标记下架 is_deleted=True
        if self.path == "/soft_delete":
            book_id = data.get("bookId", "").strip()
            find = False
            for b in book_db:
                if b["bookId"] == book_id:
                    b["is_deleted"] = True
                    find = True
                    break
            if find:
                msg = "已下架（软删除成功）"
            else:
                msg = "操作失败：没有该图书"
            self._set_headers()
            self.wfile.write(json.dumps({"msg": msg}, ensure_ascii=False).encode("utf-8"))
            return

        # 恢复上架：is_deleted=False
        if self.path == "/restore":
            book_id = data.get("bookId", "").strip()
            find = False
            for b in book_db:
                if b["bookId"] == book_id:
                    b["is_deleted"] = False
                    find = True
                    break
            if find:
                msg = "恢复上架成功"
            else:
                msg = "操作失败：没有该图书"
            self._set_headers()
            self.wfile.write(json.dumps({"msg": msg}, ensure_ascii=False).encode("utf-8"))
            return

        # 彻底删除（物理删除）
        if self.path == "/hard_delete":
            book_id = data.get("bookId", "").strip()
            idx = -1
            for i, b in enumerate(book_db):
                if b["bookId"] == book_id:
                    idx = i
                    break
            if idx != -1:
                del book_db[idx]
                msg = "彻底删除成功"
            else:
                msg = "操作失败：没有该图书"
            self._set_headers()
            self.wfile.write(json.dumps({"msg": msg}, ensure_ascii=False).encode("utf-8"))
            return

        self._set_headers(404)
        self.wfile.write(json.dumps({"msg": "路径不存在"}, ensure_ascii=False).encode("utf-8"))

if __name__ == "__main__":
    server = HTTPServer(("127.0.0.1", 8000), MyHandler)
    print("服务器启动：http://127.0.0.1:8000")
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        server.server_close()
        print("服务器关闭")
